cmake_minimum_required(VERSION 3.20)
project(pawnGO_wasm VERSION 1.0.0 LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)

# Optimization for Wasm
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -flto -DNDEBUG -pthread")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -O3 -flto -DNDEBUG -pthread")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -O3 -flto -pthread -s PTHREAD_POOL_SIZE=4 -s INITIAL_MEMORY=268435456 -s MAXIMUM_MEMORY=1073741824 -s ALLOW_MEMORY_GROWTH=1 -s WASM=1 -s ASSERTIONS=1 -s EXPORTED_RUNTIME_METHODS=['ccall','cwrap'] --preload-file ${CMAKE_SOURCE_DIR}/src/eval/nnue/nn-62ef826d1a6d.nnue@nn-62ef826d1a6d.nnue")

include_directories(
    ${CMAKE_SOURCE_DIR}/src 
    ${CMAKE_SOURCE_DIR}/src/core 
    ${CMAKE_SOURCE_DIR}/src/search 
    ${CMAKE_SOURCE_DIR}/src/eval 
    ${CMAKE_SOURCE_DIR}/src/utils 
    ${CMAKE_SOURCE_DIR}/src/eval/nnue/src 
    ${CMAKE_SOURCE_DIR}/src/eval/syzygy
)

# Do NOT define USE_AVX2, USE_BMI2, USE_POPCNT for wasm unless SIMD is explicitly used

file(GLOB_RECURSE SRC_FILES
    "src/*.cpp"
)

list(APPEND SRC_FILES "src/eval/syzygy/tbprobe.c")

# Do not include example files
list(REMOVE_ITEM SRC_FILES "${CMAKE_SOURCE_DIR}/src/eval/nnue/example.cpp")
list(REMOVE_ITEM SRC_FILES "${CMAKE_SOURCE_DIR}/src/core/magics_generator.cpp")

add_executable(pawngo_wasm
    ${SRC_FILES}
)
